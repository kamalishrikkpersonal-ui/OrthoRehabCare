# OrthoRehab-AI backend - Step 1 dependencies
# Only what is needed to read a reference video and extract MediaPipe pose landmarks.
#
# mediapipe 0.10.9 requires numpy < 2.0 (it imports numpy 1.x APIs and is not
# compatible with numpy 2.x). Therefore we pin numpy==1.26.4, which is the same
# version the upstream FitPose-Detector pins. Using a newer numpy would break
# MediaPipe at import time.

# opencv-python pinned exactly as in the upstream FitPose-Detector. Note: mediapipe
# also pulls opencv-contrib-python, so pip may resolve both opencv-python and
# opencv-contrib-python to a newer 4.x (e.g. 4.11) — that is fine with MediaPipe 0.10.9.
opencv-python==4.10.0.84
mediapipe==0.10.9
numpy==1.26.4
